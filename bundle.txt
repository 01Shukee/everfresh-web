/** Applicant-facing API. Everything except POST /applications requires the application's private token. */
export async function applicationsApi(d: VocDeps, method: string, path: string[], body: unknown, q: URLSearchParams): Promise<ApiOut> {
  const [ref, a1, a2] = path; const token = (typeof (body as any)?.token === "string" ? (body as any).token : q.get("t")) ?? "";
  if (!ref) { if (method !== "POST") return j(405, { error: "Method not allowed." }); const r = await createApplication(d, body); return r.ok ? j(r.replay ? 200 : 201, { ref: r.ref, token: r.token }) : failOut(r); }
  if (!a1) {
    if (method === "GET") { const v = await ownerView(d, ref, token); return v ? j(200, v) : j(404, { error: "Application not found." }); }
    if (method === "PUT") { const r = await updateApplication(d, ref, token, body); return r.ok ? j(200, { ok: true }) : failOut(r); }
  }
  if (a1 === "pay" && method === "POST") { const r = await startApplicationPayment(d, ref, token, (body as any)?.acknowledged === true); return r.ok ? j(200, { checkoutUrl: r.checkoutUrl, paymentStatus: r.paymentStatus }) : failOut(r); }
  if (a1 === "invoice" && a2 === "pay" && method === "POST") { const r = await startInvoicePayment(d, ref, token, (body as any)?.invoice); return r.ok ? j(200, { checkoutUrl: r.checkoutUrl }) : failOut(r); }
  if (method === "GET" && (a1 === "receipt" || a1 === "confirmation" || a1 === "invoice")) {
    const app = await appForToken(d.repo, ref, token); if (!app) return j(404, { error: "Not found." });
    if (a1 === "invoice") { const inv = (await invoicesFor(d.repo, ref)).find((i) => i.number === q.get("inv") && i.status === "issued"); return inv ? pdf(d, invoiceHtml(inv, app), `invoice-${inv.number}.pdf`) : j(404, { error: "Invoice not found." }); }
    const receipts = (await d.repo.docList<Receipt>("receipt")).filter((r) => r.appRef === ref);

/** Staff API. The caller (route) must already have verified the admin session. */
export async function adminVocApi(d: VocDeps, method: string, path: string[], body: unknown, q: URLSearchParams): Promise<ApiOut> {
  const [r, id, sub, sub2] = path; const now = d.now?.() ?? new Date();
  if (r === "applications") {
    if (!id) return method === "GET" ? j(200, { applications: await listApplications(d.repo, q.get("status") ?? undefined) }) : j(405, { error: "Method not allowed." });
    if (!sub) { if (method === "GET") { const a = await adminApplication(d.repo, id); return a ? j(200, a) : j(404, { error: "Not found." }); } if (method === "PATCH") return setApplicationStatus(d.repo, id, (body as any)?.status, (body as any)?.note); }
    if (sub === "invoices" && method === "POST") return createInvoice(d.repo, id, body, now);
  }
  if (r === "invoices" && id) {
    if (!sub && method === "PUT") return updateInvoice(d.repo, id, body, now);
    if (sub === "payments" && method === "POST") return recordManualPayment(d.repo, id, body, now);
    if (sub === "pdf" && method === "GET") { const inv = await d.repo.docGet<Invoice>("invoice", id); const app = inv ? await d.repo.docGet<Application>("app", inv.appRef) : null; return inv && app ? pdf(d, invoiceHtml(inv, app), `invoice-${id}.pdf`) : j(404, { error: "Not found." }); }
  }
  if (r === "receipts" && id && sub === "pdf" && method === "GET") { const rc = await d.repo.docGet<Receipt>("receipt", id); const app = rc ? await d.repo.docGet<Application>("app", rc.appRef) : null; return rc && app ? pdf(d, receiptHtml(rc, app), `receipt-${id}.pdf`) : j(404, { error: "Not found." }); }
  if (r === "institute") { if (method === "GET") return j(200, await getInstitute(d.repo)); if (method === "PUT") return saveInstitute(d.repo, body); }
  if (r === "content" && id) { if (method === "GET") return j(200, await getContent(d.repo, id)); if (method === "PUT") return saveContent(d.repo, id, body); }
  if (r === "media" && id && ["portfolio", "video", "testimonial"].includes(id)) {
    const kind = id as "portfolio" | "video" | "testimonial";
    if (!sub) { if (method === "GET") return j(200, { items: await listMedia(d.repo, kind) }); if (method === "POST") return saveMedia(d.repo, kind, body); }
    if (sub && method === "PUT") return saveMedia(d.repo, kind, body, sub);
    if (sub && method === "DELETE") { await d.repo.docDelete(kind, sub); return j(200, { ok: true }); }
  }
  void sub2; return j(404, { error: "Not found." });
}

export const sumKind = (l: InvoiceLine[], f: (k: LineKind) => boolean) => l.filter((x) => f(x.kind)).reduce((s, x) => s + x.amountKobo, 0);
/** Totals are always derived from lines and payments; nothing is typed in by hand. */
export function invoiceTotals(i: Pick<Invoice, "lines" | "payments">) {
  const charges = sumKind(i.lines, (k) => k !== "discount"), discounts = sumKind(i.lines, (k) => k === "discount");
  const total = charges - discounts, paid = i.payments.reduce((s, p) => s + p.amountKobo, 0);
  return { chargesKobo: charges, discountsKobo: discounts, totalKobo: total, paidKobo: paid, outstandingKobo: Math.max(total - paid, 0) };
}
export function invoiceState(i: Invoice): "draft" | "void" | "unpaid" | "part_paid" | "paid" {
  if (i.status !== "issued") return i.status; const t = invoiceTotals(i);
  return t.outstandingKobo === 0 && t.totalKobo > 0 ? "paid" : t.paidKobo > 0 ? "part_paid" : "unpaid";
}

/** Institute details shown in emails and PDFs. Editable in Admin > Vocational studies > Institute details. */
export interface Institute { name: string; address: string | null; phones: string[]; note: string | null }
export const DEFAULT_INSTITUTE: Institute = { name: "Vocational Institute", address: null, phones: [], note: null };
export async function getInstitute(repo: Repo): Promise<Institute> { return { ...DEFAULT_INSTITUTE, ...((await repo.docGet<Institute>("settings", "institute")) ?? {}) }; }
const s = (v: unknown, max: number) => (typeof v === "string" ? v.replace(/[\u0000-\u001F\u007F]/g, " ").trim().slice(0, max) : "");
export async function saveInstitute(repo: Repo, raw: unknown): Promise<{ status: number; body: unknown }> {
  const r = typeof raw === "object" && raw !== null ? (raw as Record<string, unknown>) : {}; const name = s(r.name, 80); if (!name) return { status: 422, body: { error: "Enter the institute's name." } };
  const phones = (Array.isArray(r.phones) ? r.phones : []).map((x) => s(x, 30)).filter(Boolean).slice(0, 8);
  if (phones.some((p) => !/^[+\d][\d\s()+\-]{6,}$/.test(p))) return { status: 422, body: { error: "A phone number looks wrong. Use digits, spaces, + and - only." } };
  const v: Institute = { name, address: s(r.address, 200) || null, phones, note: s(r.note, 200) || null }; await repo.docPut("settings", "institute", v); return { status: 200, body: v };

  const set = (id: string, n: number) => setQty((q) => { const next = { ...q }; if (n <= 0) delete next[id]; else next[id] = Math.min(n, 20); return next; });

  useEffect(() => { // highlight the category being read
    const ids = data.categories.map((c) => c.id); let raf = 0;
    const run = () => { cancelAnimationFrame(raf); raf = requestAnimationFrame(() => { const off = ((document.querySelector(".nav") as HTMLElement | null)?.offsetHeight ?? 72) + (bar.current?.offsetHeight ?? 0) + 24; let cur = ids[0]; for (const id of ids) { const el = document.getElementById(`vip-${id}`); if (el && el.getBoundingClientRect().top <= off) cur = id; } setActive(cur); }); };
    run(); window.addEventListener("scroll", run, { passive: true }); return () => { cancelAnimationFrame(raf); window.removeEventListener("scroll", run); };
  }, [data]);
  const go = (id: string) => { const el = document.getElementById(`vip-${id}`); if (!el) return; const off = ((document.querySelector(".nav") as HTMLElement | null)?.offsetHeight ?? 72) + (bar.current?.offsetHeight ?? 0) + 8; window.scrollTo({ top: el.getBoundingClientRect().top + window.scrollY - off, behavior: "smooth" }); setActive(id); };

  const send = async () => {
    setBusy(true); setErr("");
    try {
      const r = await fetch("/api/vip/orders", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ table, note, lines: chosen.map((i) => ({ itemId: i.id, qty: qty[i.id], note: notes[i.id] ?? "" })) }) });
      const j = await r.json().catch(() => ({})); if (!r.ok) { setErr(j.error ?? "We could not send your order. Please tell your waiter."); setBusy(false); return; }
      setDone({ ref: j.ref, url: j.whatsappUrl, totalKobo: j.totalKobo }); sheet.current?.close(); setBusy(false);
      if (j.whatsappUrl) window.location.assign(j.whatsappUrl);
    } catch { setErr("We could not reach the server. Check your connection and try again."); setBusy(false); }
  };
  const reset = () => { setQty({}); setNotes({}); setNote(""); setDone(null); setErr(""); window.scrollTo({ top: 0 }); };

  if (done) return (
    <div className="vip-done" role="status">
      <h2 className="serif">Your order is ready to send</h2>
      <p>Reference <b className="tab">{done.ref}</b> &middot; {formatNaira(done.totalKobo)}</p>
      {done.url ? <><p>WhatsApp should open with your order filled in. <b>Press Send</b> so it reaches the kitchen.</p><a className="pill" href={done.url}>Open WhatsApp</a></> : <p>Please tell your waiter your reference.</p>}
      <p><button className="linkbtn" onClick={reset}>Start a new order</button></p>
    </div>
  );
  return (
    <>
      <header className="vip-head"><span className="lab green">VIP</span><h1 className="serif">{data.title}</h1>{tableFromQr && <p className="vip-table">{data.tableLabel} {tableFromQr}</p>}{data.intro && <p className="lede">{data.intro}</p>}
        {!data.orderable && <p className="vip-note">Browse the menu and tell your waiter what you would like.</p>}</header>
      <div className="menubar" ref={bar}><div className="catrail" role="group" aria-label="Jump to category">{data.categories.map((c) => <button key={c.id} type="button" className={active === c.id ? "cat on" : "cat"} aria-current={active === c.id ? "true" : undefined} onClick={() => go(c.id)}>{c.name}</button>)}</div></div>
      {data.categories.map((c) => (
        <section key={c.id} id={`vip-${c.id}`} className="menusec" aria-labelledby={`vh-${c.id}`}>
          <h2 id={`vh-${c.id}`} className="serif">{c.name}</h2>
          <ul className="dishes">
            {c.items.map((i) => { const n = qty[i.id] ?? 0; return (
              <li key={i.id} className={`dish vip-dish ${!i.available ? "soldout" : ""}`}>
                <MediaSlot media={i.imageUrl ? { src: i.imageUrl, alt: i.imageAlt, width: 400, height: 400 } : null} label={i.name.toUpperCase()} shape="circle" className="dish-img" />
                <div className="dish-body"><h3 className="serif">{i.name}</h3>{i.description && <p>{i.description}</p>}<div className="dish-meta"><span className="price tab">{formatNaira(i.priceKobo)}</span>{!i.available && <span className="tag dark">Not available</span>}</div></div>
                {data.orderable && i.available && (n > 0
                  ? <div className="stepper qstep" role="group" aria-label={`${i.name} quantity`}><button type="button" onClick={() => set(i.id, n - 1)} aria-label={`Remove one ${i.name}`}>&minus;</button><output aria-live="polite">{n}</output><button type="button" onClick={() => set(i.id, n + 1)} disabled={n >= 20} aria-label={`Add one more ${i.name}`}>+</button></div>
                  : <button type="button" className="pill sm addbtn" onClick={() => set(i.id, 1)} aria-label={`Add ${i.name}`}><span className="addlab">Add</span><span className="plus" aria-hidden="true">+</span></button>)}
              </li>); })}
          </ul>
        </section>))}
      {data.orderable && count > 0 && <div className="vipbar"><button type="button" onClick={() => sheet.current?.showModal()}><span className="cb-count" aria-hidden="true">{count}</span><span className="cb-lab">Review order</span><span className="tab">{formatNaira(total)}</span></button></div>}
      <dialog ref={sheet} className="vipsheet" aria-labelledby="vip-rev" onClick={(e) => { if (e.target === sheet.current) sheet.current?.close(); }}>
        <div className="vipsheet-in">
          <div className="vipsheet-head"><h2 id="vip-rev" className="serif">Your order</h2><button type="button" className="dlg-x2" onClick={() => sheet.current?.close()} aria-label="Close">&times;</button></div>
          <ul className="viplines">{chosen.map((i) => (
            <li key={i.id}><div className="vl-top"><b>{i.name}</b><span className="tab">{formatNaira(i.priceKobo * qty[i.id])}</span></div>
              <div className="vl-row"><div className="stepper qstep" role="group" aria-label={`${i.name} quantity`}><button type="button" onClick={() => set(i.id, qty[i.id] - 1)} aria-label={`Remove one ${i.name}`}>&minus;</button><output>{qty[i.id]}</output><button type="button" onClick={() => set(i.id, qty[i.id] + 1)} disabled={qty[i.id] >= 20} aria-label={`Add one more ${i.name}`}>+</button></div><span className="small dim0 tab">{formatNaira(i.priceKobo)} each</span></div>
              <label className="sr" htmlFor={`n-${i.id}`}>Special instructions for {i.name}</label><input id={`n-${i.id}`} className="vl-note" placeholder="Special instructions (optional)" maxLength={200} value={notes[i.id] ?? ""} onChange={(e) => setNotes({ ...notes, [i.id]: e.target.value })} /></li>))}</ul>
          <div className="field"><label htmlFor="vip-table">{data.tableLabel}</label><input id="vip-table" value={table} maxLength={20} onChange={(e) => setTable(e.target.value)} placeholder="Table number or name" /></div>
          <div className="field"><label htmlFor="vip-note">Anything else for the kitchen? <span className="opt-t">(optional)</span></label><textarea id="vip-note" rows={2} maxLength={300} value={note} onChange={(e) => setNote(e.target.value)} /></div>
          {err && <p className="alert" role="alert">{err}</p>}
          <div className="vipsheet-foot"><div className="vl-total"><span>Total</span><b className="tab">{formatNaira(total)}</b></div><button type="button" className="pill wa" onClick={send} disabled={busy || !chosen.length}>{busy ? "Sendingâ€¦" : "Send order on WhatsApp"}</button></div>
        </div>
      </dialog>
    </>
  );
}
